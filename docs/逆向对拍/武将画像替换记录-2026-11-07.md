## 五、复现

```python
import json, os, shutil
from PIL import Image
P = r'C:/Users/lai15/Desktop/战斗系统'
cards = r'C:/Users/lai15/Desktop/rea_stzb/_all_cards_out'
m = json.load(open(os.path.join(P, 'web/data/portrait_map.json'), encoding='utf-8'))
for ourId, v in sorted(m.items()):
    src = os.path.join(cards, 'card_%d.png' % v['heroId'])
    dst = os.path.join(P, 'public/portraits', ourId + '.jpg')
    if not (os.path.exists(src) and os.path.exists(dst)):
        continue
    Image.open(src).convert('RGB').save(dst, 'JPEG', quality=92, subsampling=0, optimize=True)
```
